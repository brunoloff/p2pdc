cmake_minimum_required(VERSION 3.20)
project(p2pdc_opendht_target_test LANGUAGES NONE)
add_library(msgpack-cxx INTERFACE)
if(CASE STREQUAL "provided")
    add_library(example_opendht INTERFACE)
    set(P2PDC_OPENDHT_TARGET example_opendht)
elseif(CASE STREQUAL "missing")
    set(P2PDC_OPENDHT_TARGET missing_opendht)
else()
    message(FATAL_ERROR "unknown OpenDHT target test case")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/OpenDhtDependency.cmake")
if(NOT p2pdc_opendht_target STREQUAL "example_opendht")
    message(FATAL_ERROR "selected OpenDHT target was not preserved")
endif()
