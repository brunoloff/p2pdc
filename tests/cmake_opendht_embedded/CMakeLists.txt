cmake_minimum_required(VERSION 3.20)
project(p2pdc_embedded_target_test LANGUAGES CXX)
add_library(app_opendht INTERFACE)
add_library(msgpack-cxx INTERFACE)
if(NOT CASE STREQUAL "missing_rtc")
    add_library(app_datachannel INTERFACE)
    add_library(LibDataChannel::LibDataChannel ALIAS app_datachannel)
endif()
set(P2PDC_OPENDHT_TARGET app_opendht)
set(P2PDC_ENABLE_OPENDHT ON)
set(P2PDC_ENABLE_NATIVE_TRANSPORT ON)
set(P2PDC_ENABLE_SODIUM OFF)
set(BUILD_TESTING OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_PkgConfig ON)
set(CMAKE_DISABLE_FIND_PACKAGE_LibDataChannel ON)
set(CMAKE_DISABLE_FIND_PACKAGE_msgpack-cxx ON)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." p2pdc)
get_target_property(dependencies p2pdc LINK_LIBRARIES)
if(NOT "app_opendht" IN_LIST dependencies OR NOT "msgpack-cxx" IN_LIST dependencies OR
   NOT "LibDataChannel::LibDataChannel" IN_LIST dependencies)
    message(FATAL_ERROR "embedded library did not use the supplied dependency targets")
endif()
