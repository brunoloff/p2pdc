cmake_minimum_required(VERSION 3.20)
project(p2pdc_external_consumer LANGUAGES CXX)
set_property(GLOBAL PROPERTY JOB_POOLS consumer_pool=1)

if(NOT P2PDC_SOURCE_DIR)
    message(FATAL_ERROR "Set P2PDC_SOURCE_DIR to the p2p-discovery-connect source directory")
endif()
option(P2PDC_ANDROID_BACKEND_SMOKE "Build the Android WebRTC backend smoke test" OFF)
option(P2PDC_NATIVE_LIFECYCLE_SMOKE "Build the native node lifecycle smoke test" OFF)
option(P2PDC_NATIVE_PAIR_SMOKE "Build the native public node pair smoke test" OFF)
if(P2PDC_ANDROID_BACKEND_SMOKE AND NOT ANDROID)
    message(FATAL_ERROR "P2PDC_ANDROID_BACKEND_SMOKE requires an Android toolchain")
endif()
if(P2PDC_NATIVE_LIFECYCLE_SMOKE AND EMSCRIPTEN)
    message(FATAL_ERROR "P2PDC_NATIVE_LIFECYCLE_SMOKE requires a native target")
endif()
if(P2PDC_NATIVE_PAIR_SMOKE AND EMSCRIPTEN)
    message(FATAL_ERROR "P2PDC_NATIVE_PAIR_SMOKE requires a native target")
endif()
add_subdirectory("${P2PDC_SOURCE_DIR}" p2pdc)
add_executable(external_consumer main.cpp)
set_property(TARGET external_consumer PROPERTY JOB_POOL_COMPILE consumer_pool)
target_link_libraries(external_consumer PRIVATE p2pdc::p2pdc)
if(P2PDC_NATIVE_LIFECYCLE_SMOKE)
    add_executable(p2pdc_native_lifecycle "${P2PDC_SOURCE_DIR}/tests/native_lifecycle_smoke.cpp")
    target_link_libraries(p2pdc_native_lifecycle PRIVATE p2pdc::p2pdc)
endif()
if(P2PDC_NATIVE_PAIR_SMOKE)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(P2PDC_PAIR_DHT REQUIRED IMPORTED_TARGET opendht)
    find_package(msgpack-cxx CONFIG REQUIRED)
    add_executable(p2pdc_native_pair "${P2PDC_SOURCE_DIR}/tests/native_pair_smoke.cpp"
        "${P2PDC_SOURCE_DIR}/tests/node_connection_consumer.cpp"
        "${P2PDC_SOURCE_DIR}/tests/node_ice_validation_smoke.cpp"
        "${P2PDC_SOURCE_DIR}/tests/node_race_consumer.cpp")
    target_link_libraries(p2pdc_native_pair PRIVATE p2pdc::p2pdc PkgConfig::P2PDC_PAIR_DHT msgpack-cxx)
endif()
if(EMSCRIPTEN)
    target_link_options(external_consumer PRIVATE "-sEXIT_RUNTIME=1")
endif()
if(P2PDC_ANDROID_BACKEND_SMOKE)
    find_package(Threads REQUIRED)
    add_executable(p2pdc_android_backend_smoke
        "${P2PDC_SOURCE_DIR}/examples/native-smoke/main.cpp"
        "${P2PDC_SOURCE_DIR}/examples/native-smoke/config.cpp"
        "${P2PDC_SOURCE_DIR}/examples/native-smoke/smoke.cpp")
    target_compile_features(p2pdc_android_backend_smoke PRIVATE cxx_std_20)
    target_compile_options(p2pdc_android_backend_smoke PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(p2pdc_android_backend_smoke PRIVATE LibDataChannel::LibDataChannel Threads::Threads)
endif()
if(P2PDC_YRS_INCLUDE_DIR OR P2PDC_YRS_LIBRARY)
    if(NOT P2PDC_YRS_INCLUDE_DIR OR NOT P2PDC_YRS_LIBRARY)
        message(FATAL_ERROR "Both P2PDC_YRS_INCLUDE_DIR and P2PDC_YRS_LIBRARY are required")
    endif()
    include("${P2PDC_SOURCE_DIR}/integrations/yrs/ReferenceTarget.cmake")
endif()
