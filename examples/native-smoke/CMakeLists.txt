cmake_minimum_required(VERSION 3.20)
project(p2pdc_native_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set_property(GLOBAL PROPERTY JOB_POOLS p2pdc_heavy=1)
set(CMAKE_JOB_POOL_COMPILE p2pdc_heavy)
set(CMAKE_JOB_POOL_LINK p2pdc_heavy)
find_package(LibDataChannel CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(CTest)
add_library(smoke_config config.cpp)
target_include_directories(smoke_config PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(smoke_transport smoke.cpp)
target_link_libraries(smoke_transport PUBLIC smoke_config LibDataChannel::LibDataChannel Threads::Threads)
add_executable(native_smoke main.cpp)
target_link_libraries(native_smoke PRIVATE smoke_transport)
foreach(target smoke_config smoke_transport native_smoke)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
if(BUILD_TESTING)
    find_package(GTest CONFIG REQUIRED)
    add_executable(smoke_config_test config_test.cpp)
    target_link_libraries(smoke_config_test PRIVATE smoke_transport GTest::gtest_main)
    target_compile_options(smoke_config_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME smoke_config COMMAND smoke_config_test)
    add_test(NAME smoke_direct COMMAND native_smoke direct)
    set_tests_properties(smoke_direct PROPERTIES TIMEOUT 30 LABELS integration)
    add_test(NAME smoke_invalid_mode COMMAND native_smoke invalid)
    set_tests_properties(smoke_invalid_mode PROPERTIES WILL_FAIL TRUE)
endif()
